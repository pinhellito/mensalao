// FUNÇÃO DO RELATÓRIO DO WHATSAPP function atualizarRelatorioCobranca(filteredGames) { const textArea = document.getElementById("p-relatorio-texto"); if (!filteredGames || filteredGames.length === 0) { textArea.value = "Nenhum jogo registrado neste mês para gerar relatório."; return; } let avulsosData = {}; // id -> { qtd, total } filteredGames.forEach(jogo => { const valorDoDia = jogo.avulsoValue || 20; jogo.presentPlayers.forEach(p => { if (p.tipo === 'avulso') { if (!avulsosData[p.id]) avulsosData[p.id] = { qtd: 0, total: 0 }; avulsosData[p.id].qtd += 1; avulsosData[p.id].total += valorDoDia; } }); }); // Pega o mês selecionado e deixa em Maiúsculas (ex: AGOSTO DE 2026) const monthInput = document.getElementById("p-filtro-mes").value || currentMonthKey(); const mesRefTexto = formatMonthLong(monthInput).toUpperCase(); let txtCabecalho = `LISTA ${mesRefTexto}\nPIX: jpinhelli@gmail.com\n\n`; let txtMensal = "*Mensal (R$ 60)*\n"; // Listamos TODOS os mensais cadastrados no sistema const mensais = state.players.filter(p => p.tipo === 'mensal').sort((a,b) => a.nome.localeCompare(b.nome)); mensais.forEach(m => { txtMensal += `${m.nome} - \n`; }); let txtAvulso = "\n*Avulso (R$20/JOGO)*\n"; // Listamos apenas os avulsos que jogaram neste mês const avulsosQueJogaram = state.players.filter(p => p.tipo === 'avulso' && avulsosData[p.id]).sort((a,b) => a.nome.localeCompare(b.nome)); if(avulsosQueJogaram.length === 0) { txtAvulso += "(Nenhum avulso jogou neste mês)\n"; } else { avulsosQueJogaram.forEach(a => { const dados = avulsosData[a.id]; const txtJogos = dados.qtd === 1 ? "1 JOGO" : `${dados.qtd} JOGOS`; const valorFormat = `R$${dados.total.toFixed(2).replace('.', ',')}`; txtAvulso += `${a.nome} - ${valorFormat} (${txtJogos})\n`; }); } textArea.value = txtCabecalho + txtMensal + txtAvulso; }
